# Day 05- Advanced CI/CD and Infrastructure Provisioning

# Jenkinsfile - With Git Integration and Testing

pipeline {                                                                      # pipeline = the whole CI/CD process
    agent any                                                                   # agent any = run on any available machine
    stages {                                                                    # stages = the list of phases
        stage('Clone Repository') {                                             # stage = one phase
            steps {                                                             # steps = the commands to run in a stage
                git 'https://github.com/nikhildubey341/Nov2026_Nikhil_Pathnex'  # git = downloads code from a Git repository
            }
        }
        stage ('Build') {
            steps {
                echo 'Building project from Git Repository'                     # echo = prints the given text on the screen
            }
        }
        stage ('Test') {
            steps {
                echo 'Running unit tests'
            }
        }
        stage ('Deploy') {
            steps {
                echo 'Deploying Application'
            }
        }
    }
}

# GitLab CI/CD - With Docker Build
stages:                                                                         # stages = the order in which jobs run
  - build                                                                       # build = the stage that builds
  - push                                                                        # push = the stage that uploads
  - deploy                                                                      # deploy = the stage that releases

build:                                                                          # job name = a job GitLab runs
  stage: build                                                                  # stage: = the stage this job belongs to
  script:                                                                       # script = the commands the job runs
    - docker build -t pathnex-app .                                             # docker build = builds an image from the Dockerfile | -t = gives the image a name (tag) | . = the current folder

push:
  stage: push
  script:
    - docker push pathnex-app                                                   # docker push = uploads an image to the registry

deploy:
  stage: deploy
  script:
    - kubectl apply -f kubernetes/deployment.yaml                               # kubectl apply -f = creates or updates objects from a file

# Ansible - Create User and Set Permissions
- name: Create a new user                                                       # name (play) = a label for the whole job
  hosts: all                                                                    # hosts: all = run on every server in the inventory file
  become: yes                                                                   # become: yes = run as admin (sudo)
  tasks:                                                                        # tasks = the list of steps to run in order
    - name: Create user                                                         # name (task) = a label for one step
      user:                                                                     # user = module that manages user accounts
        name: pathnex                                                           # name (user module) = the user name
        state: present                                                          # state: present = make sure the user exists
        shell: /bin/bash                                                        # shell = the command-line program the user gets
    - name: Set permissions for the user
      file:                                                                     # file = module that manages file and folder settings
        path: /home/pathnex                                                     # path = the file or folder to change
        owner: pathnex                                                          # owner = the user who owns it
        group: pathnex                                                          # group = the group that owns it
        mode: '0755'                                                            # mode = who can read, write, run (0755)

# Terraform - EC2 with Tags and Volume
resource "aws_instance" "PathnexEC2" {                                          # resource = a thing Terraform creates (type + my own nickname)
  ami           = "ami-0abcd1234abcd1234"                                       # ami = the OS image used to create the server
  instance_type = "t2.medium"                                                   # instance_type = the size of the server (CPU and RAM)
  
  tags = {                                                                      # tags = labels used to find and organise the server
    Name = "Pathnex-Server"                                                     # Name = the name shown in the AWS console
  }

  ebs_block_device {                                                            # ebs_block_device = an extra disk attached to the server
    device_name = "/dev/sdh"                                                    # device_name = the disk's name on the server
    volume_size = 50                                                            # volume_size = the disk size in GB
  }
}

# Docker File
FROM python:3.11                                                                # FROM = the base image to start from | python:3.11 = the Python image, version 3.11

COPY app.py /opt/pathnex/python-app/                                            # COPY = copies a file from my computer into the image
CMD ["python", "/opt/pathnex/python-app/app.py"]                                # CMD = the default command that runs when the container starts | python = runs a Python file
